    PROCEDURE DefragmentRowID@5272724(Type@5077900 : Integer;Subtype@5077901 : Integer;ID@5077902 : Code[20];BatchName@5077903 : Code[20];ProdOrderLine@5077904 : Integer;RefNo@5077905 : Integer) : Text[250];
    VAR
      StrArray@5077906 : ARRAY [2] OF Text[100];
      Pos@5077907 : Integer;
      Len@5077908 : Integer;
      T@5077909 : Integer;
    BEGIN
      // Funktion zum zusammensetzen der RowID
      // somit werden alle wichtigen Kriterien generiert die fÅr einen Filteraufbau von nîten sind

      StrArray[1] := ID;
      StrArray[2] := BatchName;
      for T := 1 to 2 do begin
        if STRPOS(StrArray[T],'"') > 0 then begin
          Len := STRLEN(StrArray[T]);
          Pos := 1;
          repeat
            if COPYSTR(StrArray[T],Pos,1) = '"' then begin
              StrArray[T] := INSSTR(StrArray[T],'"',Pos+1);
              Len += 1;
              Pos += 1;
            end;
            Pos += 1;
          until Pos > Len;
        end;
      end;
      exit(STRSUBSTNO('"%1";"%2";"%3";"%4";"%5";"%6"',Type,Subtype,StrArray[1],StrArray[2],ProdOrderLine,RefNo));
    END;
